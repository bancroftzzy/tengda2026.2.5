```csharp
private void SetDefaultDuncanChangParameters()
{
    double Pa = 100.0; // 大气压 kPa（第 57 行）
    
    // 根据土层类型设置默认参数（第 59 行开始）
    if (Type.Contains("软") || Type.Contains("流"))
    {
        // 软塑粘土（第 62-67 行）
        K_Duncan = 200;
        n_Duncan = 0.4;
        Rf = 0.75;
        Kb = 100;
        m_Duncan = 0.2;
    }
    else if (Type.Contains("可塑") || Type.Contains("中"))
    {
        // 可塑粘土（第 71-76 行）
        K_Duncan = 400;
        n_Duncan = 0.5;
        Rf = 0.80;
        Kb = 200;
        m_Duncan = 0.3;
    }
    // ... 其他土类
    else  // 第 97 行
    {
        // 默认值（中等土）- 第一层土"杂填土"匹配这里
        K_Duncan = 400;
        n_Duncan = 0.5;
        Rf = 0.80;
        Kb = 200;
        m_Duncan = 0.3;
    }
    
    // 如果有Es值，用来校正K值（第 105 行）
    if (Es > 0)
    {
        // K ≈ Es / Pa × 经验系数（第 108 行）
        double estimatedK = Es * 1000 / Pa * 4.0;  
        // Es=2.12 MPa → 2120 kPa
        // estimatedK = 2120 / 100 * 4.0 = 84.8
        
        // 使用估算值和默认值的平均（第 111 行）
        K_Duncan = (K_Duncan + estimatedK) / 2.0;
        // K_Duncan = (400 + 84.8) / 2 = 242.4
    }
}
```

## 支撑轴力可调整功能详解

### 什么是"轴力可调整"？

**定义**：支撑轴力可调整（`AdjAble = True`）表示该支撑可以通过**主动调节**来优化围护结构的受力和变形。

### 工程背景

在实际基坑工程中：

1. **钢支撑**：可以通过液压千斤顶施加预应力，调整轴力
   - 优点：可调节、可拆除、施工灵活
   - 缺点：需要监测和维护

2. **混凝土支撑**：浇筑后轴力基本固定，难以调整
   - 优点：刚度大、承载力高、稳定性好
   - 缺点：不可调节、拆除困难

### 从数据文件看支撑类型

**文件**：`2-SupportsInfo.txt`

```

1	混	0.4	2.5	1	0.8	7000	2120	False    ← 混凝土，不可调
2	钢	5.2	2.5	0.609	0.016	1795	0	True     ← 钢支撑，可调
3	钢	9	2.5	0.609	0.016	1795	0	True     ← 钢支撑，可调
```

- 混凝土支撑：`AdjAble = False`（不可调整）

### 轴力调整的计算原理

**文件**：`ActiveControl/ActiveControl/Forms/Form_Calculate.cs`  
**行号**：第 99-101 行，第 583-585 行

#### 1. 记录可调支撑索引

#### 2. 轴力优化目标

**目标函数**：最小化围护结构的水平位移

```
min f(F) = ||U(F)||
```

其中：
- `F`：支撑轴力调整量向量
- `U(F)`：对应的围护结构位移

**约束条件**：
```
1. 支撑轴力限制：-MaxFC ≤ F ≤ MaxFT
2. 围护结构内力限制：-M1 ≤ M ≤ M2, |Q| ≤ Qmax
3. 变形限制：|U| ≤ Ulimit
```

#### 3. 优化算法

程序使用**粒子群优化算法（PSO）**进行轴力调整：

**文件**：`Form_Calculate.cs`  
**行号**：第 1014-1016 行

```csharp
// 第 1014 行：轴力调整过程计算及输出

mf.rtbOutputWindow.Text += "  " + "----------------------------------------------------\r\n";
```

**算法步骤**：
1. 初始化粒子群（多组轴力调整方案）
2. 对每个粒子：
   - 施加轴力调整（通过初应变实现）
   - 求解有限元方程
   - 计算目标函数值（位移范数）
   - 检查约束条件
3. 更新粒子速度和位置
4. 迭代直到收敛或达到最大迭代次数

#### 4. 初应变实现轴力调整

**原理**：通过设置支撑单元的初应变来模拟预加力

```csharp
// 施加轴力调整
Element支撑.RealConstant.IniStrn = ΔF / (E × A)
```

其中：
- `ΔF`：轴力调整量（kN）
- `E`：支撑材料弹性模量（kPa）
- `A`：支撑截面积（m²）

**等效荷载**：
```
F_等效 = E × A × ε₀ = ΔF
```

### 实际应用示例

**场景**：第2道支撑（钢支撑，可调）激活后

```csharp
    // 第 151-154 行：获取参数
    double E = Material.Emodulus;  // 弹性模量
    double A = RealConstant.Area;   // 截面面积
    double I = RealConstant.Iz;     // 惯性矩
    double L = Math.Sqrt(Math.Pow(Left.Nx - Right.Nx, 2) + 
                         Math.Pow(Left.Ny - Right.Ny, 2));
    double EA = E * A;
    double EI = E * I;
    
    // 第 158 行：检查单元长度
    if (L < 1e-10)
        MessageBox.Show("单元两端节点重合，请检查输入！", "错误");

    // 第 161-168 行：构造刚度矩阵
    double[,] tempKe = {
        {  EA/L,            0,         0, -EA/L,            0,         0 },
        {     0,  12*EI/L/L/L,  6*EI/L/L,     0, -12*EI/L/L/L,  6*EI/L/L },
        {     0,     6*EI/L/L,    4*EI/L,     0,    -6*EI/L/L,    2*EI/L },
        { -EA/L,            0,         0,  EA/L,            0,         0 },
        {     0, -12*EI/L/L/L, -6*EI/L/L,     0,  12*EI/L/L/L, -6*EI/L/L },
        {     0,     6*EI/L/L,    2*EI/L,     0,    -6*EI/L/L,    4*EI/L } };
    
    // 第 169 行：返回矩阵
    return Matrix<double>.Build.DenseOfArray(tempKe);
}
```

#### 4.2.2 单元刚度矩阵叠加

```csharp
    // 第 16 行：遍历所有单元
    foreach (Element iElem in Elements)
    {
        // 1. 计算单元全局刚度矩阵（第 19 行）
        Matrix<double> tempKg = iElem.Tr.Transpose() * iElem.Ke() * iElem.Tr;
        // tempKg: 6×6矩阵
        
        // 2. 找到单元节点在总刚矩阵中的位置（第 22-28 行）
        int starti = 0, startj = 0;
        for (int i = 0; i < Nodes.Count(); i++)
        {
            if (Nodes[i].No == iElem.Left.No)
                starti = i;  // 左节点索引
            if (Nodes[i].No == iElem.Right.No)
                startj = i;  // 右节点索引
        }
        
        // 3. 将6×6矩阵分成4块，叠加到总刚矩阵（第 31-42 行）
        
        // 左上块（3×3）：左节点与左节点的耦合
        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++)
                Kg[starti * 3 + i, starti * 3 + j] += tempKg[i, j];
        
        // 右上块（3×3）：左节点与右节点的耦合
        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++)
                Kg[starti * 3 + i, startj * 3 + j] += tempKg[i, j + 3];
        
        // 左下块（3×3）：右节点与左节点的耦合
        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++)
                Kg[startj * 3 + i, starti * 3 + j] += tempKg[i + 3, j];
        
        // 右下块（3×3）：右节点与右节点的耦合
        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++)
                Kg[startj * 3 + i, startj * 3 + j] += tempKg[i + 3, j + 3];
    }
    
    // 第 43 行：返回总刚矩阵
    return Kg;
}
```

```csharp
public static Vector<double> GetFg(List<Element> Elements, Vector<double> Fs)
{
    // 初始化初应变荷载向量
    Vector<double> Fi = Vector<double>.Build.Dense(Fs.Count());
    
    // 遍历所有单元
    foreach (Element i in Elements)
    {
        // 计算初应变等效力

                       i.RealConstant.Area * 
                       i.RealConstant.IniStrn;
        
        // 左节点：压力（负号）
        Fi[(i.Left.No - 1) * 3] += -cForce;

        Fi[(i.Right.No - 1) * 3] += cForce;
    }
    
    // 总荷载 = 外荷载 + 初应变等效荷载
    return Fs + Fi;
}
```
